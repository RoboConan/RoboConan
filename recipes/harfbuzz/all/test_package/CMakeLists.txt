cmake_minimum_required(VERSION 3.15)
project(test_package LANGUAGES C)

find_package(harfbuzz REQUIRED CONFIG)

add_executable(${PROJECT_NAME} test_package.c)
target_link_libraries(${PROJECT_NAME} PRIVATE harfbuzz::harfbuzz)
foreach(component IN ITEMS raster vector gpu)
    if(TARGET harfbuzz::${component})
        target_link_libraries(${PROJECT_NAME} PRIVATE harfbuzz::${component})
        string(TOUPPER ${component} component_upper)
        target_compile_definitions(${PROJECT_NAME} PRIVATE WITH_HARFBUZZ_${component_upper})
    endif()
endforeach()
target_compile_features(${PROJECT_NAME} PRIVATE c_std_99)
