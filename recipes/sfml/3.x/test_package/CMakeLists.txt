cmake_minimum_required(VERSION 3.15)
project(test_package LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(SFML_COMPONENTS System)
if(SFML_WITH_WINDOW)
    list(APPEND SFML_COMPONENTS Window)
endif()
if(SFML_WITH_GRAPHICS)
    list(APPEND SFML_COMPONENTS Graphics)
endif()
if(SFML_WITH_NETWORK)
    list(APPEND SFML_COMPONENTS Network)
endif()
if(SFML_WITH_AUDIO)
    list(APPEND SFML_COMPONENTS Audio)
endif()

find_package(SFML 3.1 CONFIG REQUIRED COMPONENTS ${SFML_COMPONENTS})

function(add_sfml_component_test component)
    string(TOLOWER "${component}" component_lower)
    add_executable("test_${component_lower}" test_package.cpp)
    target_compile_definitions("test_${component_lower}" PRIVATE "TEST_SFML_${component}")
    target_link_libraries("test_${component_lower}" PRIVATE "SFML::${component}")
endfunction()

add_sfml_component_test(System)
if(SFML_WITH_WINDOW)
    add_sfml_component_test(Window)
endif()
if(SFML_WITH_GRAPHICS)
    add_sfml_component_test(Graphics)
endif()
if(SFML_WITH_NETWORK)
    add_sfml_component_test(Network)
endif()
if(SFML_WITH_AUDIO)
    add_sfml_component_test(Audio)
endif()
