cmake_minimum_required(VERSION 3.26)
project(test_package C)

set(Python3_COMPONENTS Development.Module Development.Embed)
if(NOT CMAKE_CROSSCOMPILING)
    list(APPEND Python3_COMPONENTS Interpreter)
endif()
find_package(Python3 REQUIRED COMPONENTS ${Python3_COMPONENTS})

message("Python3_EXECUTABLE: ${Python3_EXECUTABLE}")
message("Python3_INTERPRETER_ID: ${Python3_INTERPRETER_ID}")
message("Python3_VERSION: ${Python3_VERSION}")
message("Python3_INCLUDE_DIRS: ${Python3_INCLUDE_DIRS}")
message("Python3_LIBRARIES: ${Python3_LIBRARIES}")

if(Python3_VERSION VERSION_GREATER_EQUAL 3.12)
    find_package(Python3 REQUIRED COMPONENTS Development.SABIModule)
    set(SOABI WITH_SOABI)
endif()

option(BUILD_MODULE "Build python module")
if(BUILD_MODULE)
    python3_add_library(spam test_module.c ${SOABI})
    if(Python3_VERSION VERSION_GREATER_EQUAL 3.15)
        python3_add_library(spam_sabi MODULE USE_SABI 3.15 WITH_SOABI test_sabi_module.c)
    endif()
    if(MSVC)
        set_target_properties(spam PROPERTIES DEBUG_POSTFIX "_d")
        if(TARGET spam_sabi)
            set_target_properties(spam_sabi PROPERTIES DEBUG_POSTFIX "_d")
        endif()
    endif()
endif()

add_executable(${PROJECT_NAME} test_package.c)
target_link_libraries(${PROJECT_NAME} PRIVATE Python3::Python)
