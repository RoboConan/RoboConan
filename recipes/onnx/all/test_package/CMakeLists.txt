cmake_minimum_required(VERSION 3.15)
project(test_package LANGUAGES CXX)

find_package(ONNX REQUIRED CONFIG)

add_executable(${PROJECT_NAME} test_package.cpp)
if(TEST_ONNX_NAMESPACED_TARGETS)
    target_link_libraries(${PROJECT_NAME} PRIVATE ONNX::onnx)
else()
    target_link_libraries(${PROJECT_NAME} PRIVATE onnx)
endif()
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_11)

add_executable(${PROJECT_NAME}_proto test_package_proto.cpp)
if(TEST_ONNX_NAMESPACED_TARGETS)
    target_link_libraries(${PROJECT_NAME}_proto PRIVATE ONNX::onnx_proto)
else()
    target_link_libraries(${PROJECT_NAME}_proto PRIVATE onnx_proto)
endif()
target_compile_features(${PROJECT_NAME}_proto PRIVATE cxx_std_11)
