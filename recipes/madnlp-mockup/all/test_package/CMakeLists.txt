cmake_minimum_required(VERSION 3.15)
project(test_package LANGUAGES CXX)

add_executable(${PROJECT_NAME} test_package.cpp)
if(TEST_MADNLP_LEGACY)
  find_package(madnlp REQUIRED CONFIG)
  target_compile_definitions(${PROJECT_NAME} PRIVATE TEST_MADNLP_LEGACY)
  target_link_libraries(${PROJECT_NAME} PRIVATE madnlp::madnlp_c)
else()
  find_package(libmad REQUIRED CONFIG)
  target_link_libraries(${PROJECT_NAME} PRIVATE libmad::libMad)
endif()
