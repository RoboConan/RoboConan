cmake_minimum_required(VERSION 3.15)
project(test_package LANGUAGES C)

if(TEST_SLICOT_INTEGER8)
    find_package(SLICOT64 REQUIRED CONFIG)
elseif(NOT TARGET SLICOT::slicot)
    find_package(SLICOT QUIET CONFIG)
endif()
if(NOT TARGET SLICOT::slicot)
    find_package(slicot REQUIRED CONFIG)
endif()

add_executable(${PROJECT_NAME} test_package.c)
if(TARGET SLICOT::slicot)
    target_link_libraries(${PROJECT_NAME} PRIVATE SLICOT::slicot)
else()
    target_link_libraries(${PROJECT_NAME} PRIVATE slicot::slicot)
endif()
